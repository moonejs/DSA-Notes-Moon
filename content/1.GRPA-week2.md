```python
def combinationSort(L):
    L1 = sorted(L, key=lambda x: x[0])
    
    L2 = L1.copy()
    
    i = 0
    while i < len(L2):
        j = i
        while j < len(L2) and L2[j][0] == L2[i][0]:
            j += 1
        
        L2[i:j] = sorted(L2[i:j], key=lambda x: int(x[1:]), reverse=True)
        i = j
    
    return L1, L2
```

```python
def findLargest(L):
    st = 0
    ed = len(L) - 1
    
    while st <= ed:
        mid = (st + ed) // 2
        
        if L[st] < L[mid]:
            st = mid
        elif L[st] > L[mid]:
            ed = mid - 1
        else:
            st = mid + 1
    
    return L[ed]
```

```python
def mergeInPlace(A, B):
    i = 0
    j = 0
    
    while i < len(B):
        if j == len(A):
            j = 0
            i += 1
        elif B[i] < A[j]:
            A.swap(j, B, i)
            j += 1
    
    for i in range(len(B)):
        mn = i
        for j in range(i + 1, len(B)):
            if B[j] < B[mn]:
                mn = j
        B.swap(i, B, mn)
```
